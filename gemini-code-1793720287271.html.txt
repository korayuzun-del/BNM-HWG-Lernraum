  <script>
    const USERS = ["Koray", "Moritz", "Tom", "Marc", "Hazal"];
    const SUBJECTS = [
      { id: "recht", name: "Recht", cycle: "Jede Woche" },
      { id: "bwl", name: "Allgemeine BWL", cycle: "Alle 2 Wochen (ungerade KW)" }
    ];

    let currentUser = localStorage.getItem("bnm_active_user") || null;
    let selectedUserForAuth = null;

    // App Initialisierung
    document.addEventListener("DOMContentLoaded", () => {
      initAuthModal();
      renderMatrix();
      renderRooms();
      
      if (currentUser) {
        document.getElementById("auth-modal").classList.add("hidden");
        document.getElementById("current-user-display").innerText = "Eingeloggt als: " + currentUser;
      }
    });

    function switchTab(tab) {
      document.getElementById("view-stundenplan").classList.add("hidden");
      document.getElementById("view-links").classList.add("hidden");
      
      document.getElementById("tab-stundenplan").className = "px-3 py-1.5 rounded-lg font-medium text-slate-400 hover:text-white transition";
      document.getElementById("tab-links").className = "px-3 py-1.5 rounded-lg font-medium text-slate-400 hover:text-white transition";

      document.getElementById("view-" + tab).classList.remove("hidden");
      document.getElementById("tab-" + tab).className = "px-3 py-1.5 rounded-lg font-medium transition bg-indigo-600 text-white";
    }

    function initAuthModal() {
      const list = document.getElementById("user-select-list");
      list.innerHTML = "";
      USERS.forEach(user => {
        const hasPw = !!localStorage.getItem("bnm_pw_" + user);
        const btn = document.createElement("button");
        btn.className = "w-full bg-slate-800 hover:bg-slate-700 border border-slate-700/70 p-3 rounded-xl flex justify-between items-center text-sm font-medium transition";
        btn.onclick = () => selectUser(user);
        btn.innerHTML = `<span>${user}</span> ${hasPw ? '<i class="fa-solid fa-lock text-xs text-amber-400"></i>' : '<span class="text-xs text-slate-500">Neu</span>'}`;
        list.appendChild(btn);
      });
    }

    function selectUser(user) {
      selectedUserForAuth = user;
      document.getElementById("user-select-list").classList.add("hidden");
      document.getElementById("password-form").classList.remove("hidden");
      document.getElementById("selected-user-name").innerText = "Ausgewählt: " + user;
      
      const hasPw = !!localStorage.getItem("bnm_pw_" + user);
      if (!hasPw) {
        document.getElementById("pw-confirm-container").classList.remove("hidden");
      } else {
        document.getElementById("pw-confirm-container").classList.add("hidden");
      }
    }

    function cancelUserSelect() {
      selectedUserForAuth = null;
      document.getElementById("user-select-list").classList.remove("hidden");
      document.getElementById("password-form").classList.add("hidden");
      document.getElementById("auth-error").classList.add("hidden");
    }

    function handleAuthSubmit(e) {
      e.preventDefault();
      const pw = document.getElementById("auth-password").value;
      const confirmPw = document.getElementById("auth-password-confirm").value;
      const storedPw = localStorage.getItem("bnm_pw_" + selectedUserForAuth);

      if (!storedPw) {
        if (pw !== confirmPw) {
          showAuthError("Passwörter stimmen nicht überein!");
          return;
        }
        localStorage.setItem("bnm_pw_" + selectedUserForAuth, pw);
      } else if (storedPw !== pw) {
        showAuthError("Falsches Passwort!");
        return;
      }

      currentUser = selectedUserForAuth;
      localStorage.setItem("bnm_active_user", currentUser);
      document.getElementById("current-user-display").innerText = "Eingeloggt als: " + currentUser;
      document.getElementById("auth-modal").classList.add("hidden");
      renderMatrix();
    }

    function showAuthError(msg) {
      const err = document.getElementById("auth-error");
      err.innerText = msg;
      err.classList.remove("hidden");
    }

    function openProfileModal() {
      cancelUserSelect();
      initAuthModal();
      document.getElementById("auth-modal").classList.remove("hidden");
    }

    function renderMatrix() {
      const tbody = document.getElementById("matrix-tbody");
      tbody.innerHTML = "";

      SUBJECTS.forEach(sub => {
        const myStatus = localStorage.getItem(`status_${sub.id}_${currentUser}`) || "unsicher";
        
        const tr = document.createElement("tr");
        tr.innerHTML = `
          <td class="p-3 font-semibold text-white">${sub.name}</td>
          <td class="p-3 text-xs text-slate-400">${sub.cycle}</td>
          <td class="p-3">
            <select onchange="updateStatus('${sub.id}', this.value)" class="bg-slate-800 border border-slate-700 text-xs rounded-lg p-1.5 text-white focus:outline-none">
              <option value="sicher" ${myStatus === 'sicher' ? 'selected' : ''}>🟢 Sicher</option>
              <option value="unsicher" ${myStatus === 'unsicher' ? 'selected' : ''}>🔴 Unsicher</option>
            </select>
          </td>
          <td class="p-3 text-xs text-slate-400">Live-Sync aktiv</td>
        `;
        tbody.appendChild(tr);
      });
    }

    function updateStatus(subId, val) {
      if (!currentUser) return;
      localStorage.setItem(`status_${subId}_${currentUser}`, val);
    }

    function renderRooms() {
      const container = document.getElementById("rooms-container");
      container.innerHTML = `
        <div class="bg-slate-800/40 p-3 rounded-xl border border-slate-700/50 flex justify-between items-center">
          <div>
            <p class="font-semibold text-sm text-white">Raum 1: Mathe / Recht</p>
            <p class="text-xs text-slate-400">0 Teilnehmer</p>
          </div>
          <button class="px-3 py-1 bg-indigo-600/80 hover:bg-indigo-600 text-xs rounded-lg text-white">Beitreten</button>
        </div>
        <div class="bg-slate-800/40 p-3 rounded-xl border border-slate-700/50 flex justify-between items-center">
          <div>
            <p class="font-semibold text-sm text-white">Raum 2: BWL Lounge</p>
            <p class="text-xs text-slate-400">0 Teilnehmer</p>
          </div>
          <button class="px-3 py-1 bg-indigo-600/80 hover:bg-indigo-600 text-xs rounded-lg text-white">Beitreten</button>
        </div>
      `;
    }
  </script>
